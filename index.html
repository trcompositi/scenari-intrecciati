// ========================================== // SEZIONE 4: GESTIONE DATI E STATO // ========================================== function saveGameState() { var state = { currentPlayer: currentPlayer, currentScenarioId: currentScenarioId, activeCategories: activeCategories, currentLevelIndex: currentLevelIndex, currentDialogueIndex: currentDialogueIndex, currentScreenId: currentScreenId, selectedVictim: selectedVictim, currentDrawnAction: currentDrawnAction, isCardFlipped: isCardFlipped, isSingleDevice: isSingleDevice }; localStorage.setItem("ScenariGameState_v10", JSON.stringify(state)); } function checkSavedGameAvailable() { var saved = localStorage.getItem("ScenariGameState_v10"); if (saved) { try { var state = JSON.parse(saved); if ((state.currentPlayer || state.isSingleDevice) && state.currentScenarioId) { document.getElementById('btn-resume-game').style.display = 'inline-block'; document.getElementById('btn-reset-game').style.display = 'inline-block'; } } catch(e){} } } function resumeSavedGame() { restoreGameState(); } function restoreGameState() { var saved = localStorage.getItem("ScenariGameState_v10"); if (!saved) return false; try { var state = JSON.parse(saved); if (state.currentPlayer || state.isSingleDevice) { currentPlayer = state.currentPlayer || ""; currentScenarioId = state.currentScenarioId; activeCategories = state.activeCategories || []; currentLevelIndex = state.currentLevelIndex || 0; currentDialogueIndex = state.currentDialogueIndex || 0; currentScreenId = state.currentScreenId || "selection-screen"; selectedVictim = state.selectedVictim || ""; currentDrawnAction = state.currentDrawnAction || null; isCardFlipped = state.isCardFlipped || false; isSingleDevice = state.isSingleDevice || false; if (!isSingleDevice) { document.getElementById('btn-select-tiziano').disabled = (currentPlayer === 'Claudia'); document.getElementById('btn-select-claudia').disabled = (currentPlayer === 'Tiziano'); autoConnectP2P(currentPlayer); } document.getElementById('btn-reset-game').style.display = 'inline-block'; if (currentScenarioId && roleplayScenarios[currentScenarioId]) { var levelData = roleplayScenarios[currentScenarioId].levels[currentLevelIndex]; if (levelData) { currentLevelActions["Claudia"] = levelData.actions.filter(function(a) { return a.actor === "Claudia" && activeCategories.indexOf(a.category) !== -1; }); currentLevelActions["Tiziano"] = levelData.actions.filter(function(a) { return a.actor === "Tiziano" && activeCategories.indexOf(a.category) !== -1; }); } showScreen(currentScreenId); var isGaming = (currentScreenId === 'vn-screen' || currentScreenId === 'minigame-screen' || currentScreenId === 'cards-screen'); if (isGaming) { document.getElementById('game-menu-btn').style.display = 'block'; document.getElementById('fixed-skip-level-btn').style.display = 'block'; } if (currentScreenId === 'vn-screen') { loadLevelNarrative(); } else if (currentScreenId === 'minigame-screen') { openMinigame(true); } else if (currentScreenId === 'cards-screen') { goToCards(true); if (isCardFlipped) revealCard(true); } } else { showScreen('selection-screen'); } return true; } } catch(e) { console.error("Errore ripristino stato:", e); } return false; } function manualSaveGame() { saveGameState(); alert("💾 Partita salvata con successo!"); closePauseModal(); } function abandonGame() { if (confirm("Vuoi davvero abbandonare la partita attuale e tornare al menu iniziale?")) { closePauseModal(); showScreen('selection-screen'); } } function resetGameState() { if (confirm("Vuoi azzerare la partita attuale per iniziarne una nuova?")) { localStorage.removeItem("ScenariGameState_v10"); location.reload(); } }
